\begin{proposition}[Identity]\label{prop:id}
  For every $n \times n$ matrix $A$ we have $I_n A = A$.
\end{proposition}
\begin{exercise}\label{ex:hess}
  Compute $\hessian[f][x][]{2}$ for $f(x_1, x_2) = x_1^2 x_2$.
\end{exercise}
By \proposition{prop:id}, \exercise{ex:hess} needs no identity matrix.
